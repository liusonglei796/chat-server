## 2. Handler 实现

```go
// RecallMessage 撤回消息
// POST /messages/recall
// 安全: 从JWT上下文获取当前用户ID，Service层校验发送者身份
func (h *MessageHandler) RecallMessage(c *gin.Context) {
    userId, exists := c.Get("user_id")
    if !exists {
        HandleError(c, errorx.New(errorx.CodeUnauthorized, "请先登录"))
        return
    }

    var req message.RecallMessageRequest
    if err := c.ShouldBindJSON(&req); err != nil {
        HandleParamError(c, err)
        return
    }

    if err := h.messageSvc.RecallMessage(userId.(string), req); err != nil {
        HandleError(c, err)
        return
    }
    HandleSuccess(c, nil)
}
```

### 3.2 实现逻辑

```go
// RecallMessage 撤回消息
// 业务规则：
// 1. 只能撤回自己发送的消息
// 2. 发送时间在 2 分钟内可撤回
func (s *messageService) RecallMessage(userId string, req RecallMessageRequest) error {
    // 1. 查询消息
    msg, err := s.repos.Message.FindById(req.MessageId)
    if err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) {
            return errorx.New(errorx.CodeNotFound, "消息不存在")
        }
        return errorx.WrapDBError(err, "查询消息失败")
    }

    // 2. 校验发送者身份（只能撤回自己发的消息）
    if msg.SenderId != userId {
        return errorx.New(errorx.CodeNoPermission, "只能撤回自己发送的消息")
    }

    // 3. 检查撤回时间限制（2分钟）
    recallTimeLimit := 2 * time.Minute
    if time.Since(msg.CreatedAt) > recallTimeLimit {
        return errorx.New(errorx.CodeInvalidParam, "消息已超过撤回时限")
    }

    // 4. 更新消息状态为已撤回
    if err := s.repos.Message.UpdateRecallStatus(req.MessageId); err != nil {
        return errorx.WrapDBError(err, "更新消息状态失败")
    }

    // 5. 异步清理相关缓存
    go func() {
        s.cache.Del(context.Background(), fmt.Sprintf("message:%d", req.MessageId))
    }()

    return nil
}
```

## 4. DAO 层

### 4.1 查询消息

```go
// FindById 根据ID查询消息
func (r *messageRepository) FindById(id uint) (*model.Message, error) {
    var msg model.Message
    if err := r.db.First(&msg, id).Error; err != nil {
        return nil, wrapDBError(err, "查询消息")
    }
    return &msg, nil
}
```

### 4.2 更新撤回状态

```go
// UpdateRecallStatus 更新消息为已撤回状态
func (r *messageRepository) UpdateRecallStatus(messageId uint) error {
    result := r.db.Model(&model.Message{}).
        Where("id = ?", messageId).
        Updates(map[string]interface{}{
            "is_recall": true,  // 已撤回标记
            "content":    "",   // 清空内容
        })
    if result.Error != nil {
        return wrapDBError(result.Error, "更新撤回状态")
    }
    return nil
}
```

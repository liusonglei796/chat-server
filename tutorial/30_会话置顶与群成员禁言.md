## 1. 会话置顶

### 1.1 接口定义

| 接口 | 方法 | 路径 | 说明 |
|-----|------|------|------|
| 置顶/取消置顶会话 | PUT | `/sessions/pin` | 设置或取消会话置顶 |

### 请求参数

```go
// PinSessionRequest 置顶会话请求
type PinSessionRequest struct {
    SessionId uint `json:"sessionId" binding:"required"` // 会话ID
    IsPinned  bool `json:"isPinned"`                   // true: 置顶, false: 取消置顶
}
```

### 1.2 Handler 实现

```go
// PinSession 置顶/取消置顶会话
// PUT /sessions/pin
// 安全: 从JWT上下文获取当前用户ID，Service层校验会话归属
func (h *SessionHandler) PinSession(c *gin.Context) {
    userId, exists := c.Get("user_id")
    if !exists {
        HandleError(c, errorx.New(errorx.CodeUnauthorized, "请先登录"))
        return
    }

    var req session.PinSessionRequest
    if err := c.ShouldBindJSON(&req); err != nil {
        HandleParamError(c, err)
        return
    }

    if err := h.sessionSvc.PinSession(userId.(string), req.SessionId, req.IsPinned); err != nil {
        HandleError(c, err)
        return
    }
    HandleSuccess(c, nil)
}
```

### 1.3 Service 实现

```go
// PinSession 置顶/取消置顶会话
func (s *sessionService) PinSession(userId string, sessionId uint, isPinned bool) error {
    // 1. 校验会话归属
    session, err := s.repos.Session.FindById(sessionId)
    if err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) {
            return errorx.New(errorx.CodeNotFound, "会话不存在")
        }
        return errorx.WrapDBError(err, "查询会话失败")
    }

    // 2. 校验会话是否属于当前用户
    if session.UserId != userId {
        return errorx.New(errorx.CodeNoPermission, "无权限操作此会话")
    }

    // 3. 更新置顶状态
    if err := s.repos.Session.UpdatePinned(sessionId, isPinned); err != nil {
        return errorx.WrapDBError(err, "更新置顶状态失败")
    }

    // 4. 如果是置顶操作，需要更新排序时间
    if isPinned {
        if err := s.repos.Session.UpdateActiveTime(sessionId); err != nil {
            return errorx.WrapDBError(err, "更新会话活跃时间失败")
        }
    }

    // 5. 清理缓存
    go func() {
        s.cache.Del(context.Background(), fmt.Sprintf("session:%d", sessionId))
    }()

    return nil
}
```

```go
// MuteMember 禁言/取消禁言群成员
func (g *groupInfoService) MuteMember(operatorId string, req group.MuteMemberRequest) error {
    // 1. 校验群组是否存在
    groupInfo, err := g.repos.Group.FindByUuid(req.GroupId)
    if err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) {
            return errorx.New(errorx.CodeNotFound, "群组不存在")
        }
        return errorx.WrapDBError(err, "查询群组失败")
    }

    // 2. 校验操作者权限（群主或管理员）
    member, err := g.repos.Member.FindMember(req.GroupId, operatorId)
    if err != nil {
        return errorx.New(errorx.CodeNoPermission, "无权限操作")
    }
    if member.Role != group_role_enum.OWNER && member.Role != group_role_enum.ADMIN {
        return errorx.New(errorx.CodeNoPermission, "只有群主和管理员可以禁言")
    }

    // 3. 校验被禁言成员是否存在
    targetMember, err := g.repos.Member.FindMember(req.GroupId, req.UserId)
    if err != nil {
        return errorx.New(errorx.CodeNotFound, "群成员不存在")
    }

    // 4. 群主不能被禁言
    if targetMember.Role == group_role_enum.OWNER {
        return errorx.New(errorx.CodeNoPermission, "群主不能被禁言")
    }

    // 5. 更新禁言状态
    var muteEndTime *time.Time
    if req.Duration > 0 {
        endTime := time.Now().Add(time.Duration(req.Duration) * time.Minute)
        muteEndTime = &endTime
    }

    if err := g.repos.Member.UpdateMuteStatus(req.UserId, req.GroupId, muteEndTime); err != nil {
        return errorx.WrapDBError(err, "更新禁言状态失败")
    }

    return nil
}
```

## 3. 禁言检查

在发送消息时需要检查成员是否被禁言：

```go
// CheckMuteStatus 检查是否被禁言
func (g *groupInfoService) CheckMuteStatus(userId, groupId string) error {
    member, err := g.repos.Member.FindMember(groupId, userId)
    if err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) {
            return errorx.New(errorx.CodeNotFound, "不是群成员")
        }
        return errorx.WrapDBError(err, "查询成员失败")
    }

    // 检查是否在禁言期
    if member.MuteEndTime != nil && time.Now().Before(*member.MuteEndTime) {
        remaining := time.Until(*member.MuteEndTime)
        return errorx.Newf(errorx.CodeNoPermission, "你已被禁言，剩余时间: %d 分钟", int(remaining.Minutes()))
    }

    return nil
}
```

| 批量设置群组状态 | POST | `/admin/group/setStatus` | 批量启用/禁用群组 |

## 2. 路由注册

### admin_routes.go

```go
// RegisterAdminRoutes 注册管理员相关路由（需要认证 + 管理员权限）
func (rt *Router) RegisterAdminRoutes(rg *gin.RouterGroup) {
    adminGroup := rg.Group("/admin")
    // 应用管理员权限校验中间件（双重校验：JWT Claims + 实时查库）
    adminGroup.Use(middleware.AdminAuth(rt.adminChecker))
    {
        // ===== 用户管理 =====
        userAdminGroup := adminGroup.Group("/user")
        {
            userAdminGroup.GET("/list", rt.handlers.Admin.GetUserListPaged)
            userAdminGroup.POST("/setAdmin", rt.handlers.Admin.SetAdmin)
            userAdminGroup.POST("/batchStatus", rt.handlers.Admin.BatchUpdateUserStatus)
        }

        // ===== 群组管理 =====
        groupAdminGroup := adminGroup.Group("/group")
        {
            groupAdminGroup.GET("/list", rt.handlers.Admin.GetGroupInfoList)
            groupAdminGroup.POST("/delete", rt.handlers.Admin.DeleteGroups)
            groupAdminGroup.POST("/setStatus", rt.handlers.Admin.SetGroupsStatus)
        }
    }
}
```

## 3. 权限验证

### AdminAuth 中间件

```go
// AdminAuth 管理员权限校验中间件
// 双重校验：JWT Claims 快速拒绝 + 实时查库校验
func AdminAuth(adminChecker func(string) bool) gin.HandlerFunc {
    return func(c *gin.Context) {
        // 1. 从 JWT 获取用户 UUID
        uuid := c.GetString("userUuid")
        if uuid == "" {
            c.JSON(401, gin.H{"ret": -1, "msg": "未登录"})
            c.Abort()
            return
        }

        // 2. 快速校验：JWT Claims 中的 isAdmin 字段
        isAdmin := c.GetInt("isAdmin")
        if isAdmin != 1 {
            c.JSON(403, gin.H{"ret": -1, "msg": "无权限"})
            c.Abort()
            return
        }

        // 3. 可选：实时查库校验（防止权限撤销后仍可访问）
        if adminChecker != nil && !adminChecker(uuid) {
            c.JSON(403, gin.H{"ret": -1, "msg": "权限已撤销"})
            c.Abort()
            return
        }

        c.Next()
    }
}
```

## 6. Service 层实现

### AdminService

```go
type adminService struct {
    repos *mysql.Repositories
    cache myredis.CacheService
}

func NewAdminService(repos *mysql.Repositories, cache myredis.CacheService) *adminService {
    return &adminService{repos: repos, cache: cache}
}

// 批量设置管理员
func (s *adminService) SetAdmin(uuids []string, isAdmin int8) error {
    if len(uuids) == 0 {
        return nil
    }
    if err := s.repos.User.UpdateUserIsAdminByUuids(uuids, isAdmin); err != nil {
        return errorx.ErrServerBusy
    }
    // 异步清理缓存
    go func() {
        for _, uuid := range uuids {
            myredis.DelKeyIfExists("user_info_" + uuid)
        }
    }()
    return nil
}

// 批量更新用户状态
func (s *adminService) BatchUpdateUserStatus(req request.BatchUpdateUserStatusRequest) error {
    if len(req.Uuids) == 0 {
        return nil
    }
    err := s.repos.Transaction(func(txRepos *mysql.Repositories) error {
        if err := txRepos.User.UpdateUserStatusByUuids(req.Uuids, req.Status); err != nil {
            return errorx.ErrServerBusy
        }
        // 软删除相关会话
        if err := txRepos.Session.SoftDeleteByUsers(req.Uuids); err != nil {
            return errorx.ErrServerBusy
        }
        return nil
    })
    return err
}
```
